```

  → INSERT assignment
```

### 11.3 api/assign.php

```php
<?php
require_role(['super', 'admin']);
csrf_validate();

$roomId = $_POST['room_id'];
$agentId = $_POST['agent_id'];

// Business rules
if (AssignmentRepository::hasActiveAssignment($roomId)) {
    http_response_code(409);
    echo json_encode(['error' => 'Room already assigned']);
    exit;
}

$activeCount = AssignmentRepository::countActiveForAgent($agentId);
$max = AgentRepository::getMaxConcurrent($agentId);
if ($activeCount >= $max) {
    http_response_code(422);
    echo json_encode(['error' => 'Agent at max concurrent limit']);
    exit;
}

AssignmentRepository::assign($roomId, $agentId, $_SESSION['user_id']);
audit_log('assignment.create', "chat_room_assignments/{$roomId}");
echo json_encode(['success' => true]);
```

### 12.6 Performance stats query

```sql
SELECT a.display_name,
       COUNT(DISTINCT CASE WHEN cr.status='closed' THEN cr.id END) AS closed_today,
       AVG(TIMESTAMPDIFF(SECOND, cr.created_at, cr.closed_at)) AS avg_resolution
FROM agents a
LEFT JOIN chat_rooms cr ON cr.agent_id = a.id AND DATE(cr.closed_at)=CURDATE()
WHERE a.deleted_at IS NULL
GROUP BY a.id;
```

### 12.8 chat_room_assignments DDL

```sql
CREATE TABLE chat_room_assignments (
    id           VARCHAR(36)  NOT NULL PRIMARY KEY,
    room_id      VARCHAR(36)  NOT NULL,
    agent_id     VARCHAR(36)  NOT NULL,
    assigned_at  DATETIME(3)  NOT NULL DEFAULT CURRENT_TIMESTAMP(3),
    released_at  DATETIME(3)  NULL,
    assigned_by  VARCHAR(36)  NULL,
    note         VARCHAR(255) NULL,
    CONSTRAINT fk_cra_room FOREIGN KEY (room_id) REFERENCES chat_rooms(id),
    CONSTRAINT fk_cra_agent FOREIGN KEY (agent_id) REFERENCES agents(id),
    INDEX idx_cra_agent_active (agent_id, released_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
```
